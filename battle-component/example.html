🎮 Pokemon Battle Component Demo

Interactive Battle

<pokemon-battle id="battle" interactive>
</pokemon-battle>

<script>
  const battle = document.getElementById('battle');

  // Listen for choice requests
  battle.addEventListener('choice-request', (e) => {
    console.log('Request:', e.detail);
  });

  // Listen for choices made
  battle.addEventListener('choice-made', (e) => {
    console.log('Choice:', e.detail.complete);
  });

  // Make a choice
  battle.makeChoice('move 1');
</script>
